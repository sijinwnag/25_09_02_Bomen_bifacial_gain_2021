print("\n=== CONVERTING HOURLY TO DAILY ENERGY FOR METRICS ===")

def hourly_to_daily_clip_only(hourly_power):
    """Clip hourly power (if enabled) and aggregate to daily energy with proper handling of missing values."""
    if apply_clipping and clipping_threshold is not None:
        clipped_hourly = hourly_power.clip(upper=clipping_threshold)
    else:
        clipped_hourly = hourly_power
    
    # Count valid hours per day
    valid_hours_per_day = clipped_hourly.resample('D').count()
    
    # Sum the daily energy
    daily_sum = clipped_hourly.resample('D').sum()
    
    # Only keep days with at least 12 valid hours (adjust threshold as needed)
    # For other days, set to NaN (will not be connected in plot)
    valid_days = valid_hours_per_day >= 12
    daily_sum_filtered = daily_sum.where(valid_days)
    
    return daily_sum_filtered
