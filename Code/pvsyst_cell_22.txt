# Aggregate hourly power to daily energy for optimization
def hourly_to_daily_for_opt(hourly_power, min_hours=12):
    """Convert hourly power to daily energy, requiring minimum valid hours per day."""
    # Count valid hours per day
    valid_hours_per_day = hourly_power.resample('D').count()
    
    # Sum the daily energy
    daily_sum = hourly_power.resample('D').sum()
    
    # Only keep days with at least min_hours valid hours
    valid_days = valid_hours_per_day >= min_hours
    daily_sum_filtered = daily_sum.where(valid_days)
    
    return daily_sum_filtered

# Create daily actual energy for optimization
daily_actual_for_opt = hourly_to_daily_for_opt(hourly_metrics_df['Actual'])

# Function to calculate MBE for a given scale factor
def calculate_mbe(scale_factor):
    """Calculate Mean Bias Error for given scale factor."""
    # Scale the simulated data 
    scaled_sim = hourly_metrics_df['Simulated'] * scale_factor
    
    # Apply clipping if enabled
    if apply_clipping:
        scaled_sim = scaled_sim.clip(upper=clipping_threshold)
    
    # Convert to daily energy
    daily_sim = hourly_to_daily_for_opt(scaled_sim)
    
    # Filter to include only days where both actual and simulated exist
    common_idx = daily_actual_for_opt.dropna().index.intersection(daily_sim.dropna().index)
    
    if len(common_idx) == 0:
        return float('inf')
    
    # Calculate MBE on daily values
    return np.mean(daily_sim[common_idx] - daily_actual_for_opt[common_idx])

# Binary search to find optimal scaling factor
def find_optimal_scaling_factor(min_factor=0.5, max_factor=1.5, max_iterations=100):
    print("\nStarting binary search for optimal scaling factor...")
    print(f"Initial search range: [{min_factor:.8f}, {max_factor:.8f}]")
    
    iterations = 0
    best_factor = None
    best_mbe = float('inf')
    
    while iterations < max_iterations:
        iterations += 1
        mid_factor = (min_factor + max_factor) / 2
        
        # Calculate MBE for the current factor
        mbe = calculate_mbe(mid_factor)
        
        # Track the best factor found
        if abs(mbe) < abs(best_mbe):
            best_factor = mid_factor
            best_mbe = mbe
        
        print(f"Iteration {iterations}: Factor = {mid_factor:.10f}, MBE = {mbe:.10e}")
        
        # Check if we've reached the target precision
        if abs(mbe) < target_mbe_tolerance:
            print(f"✓ Converged! Found factor with MBE below tolerance.")
            return mid_factor, mbe, iterations
        
        # Adjust search range based on MBE sign
        if mbe > 0:  # MBE is positive, need to decrease factor
            max_factor = mid_factor
        else:  # MBE is negative, need to increase factor
            min_factor = mid_factor
        
        # Check if search range is too small (reached numerical precision limit)
        if max_factor - min_factor < 1e-15:
            print(f"! Reached numerical precision limit after {iterations} iterations.")
            print(f"  Best MBE achieved: {best_mbe:.12e}")
            return best_factor, best_mbe, iterations
    
    print(f"! Reached maximum iterations ({max_iterations}).")
    print(f"  Best MBE achieved: {best_mbe:.12e}")
    return best_factor, best_mbe, iterations

    return daily_sum_filtered
